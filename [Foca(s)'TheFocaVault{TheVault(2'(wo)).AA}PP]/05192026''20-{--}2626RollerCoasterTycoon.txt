; ==============================================================================
; ROLLERCOASTER TYCOON SIMULATION CONCEPT: UPDATING GUEST STATS
; ==============================================================================

.386                           ; Enable Intel 80386 instructions (32-bit registers)
.MODEL FLAT, C                 ; Use flat memory model
.DATA

; Define a structured record for a single park guest
GUEST_STRUCT STRUC
    GuestID    DW ?            ; 2 bytes: Unique ID
    PosX       DW ?            ; 2 bytes: X tile coordinate
    PosY       DW ?            ; 2 bytes: Y tile coordinate
    Cash       DW ?            ; 2 bytes: Cash remaining (in cents)
    Nausea     DB ?            ; 1 byte:  Nausea level (0 - 255)
    Happiness  DB ?            ; 1 byte:  Happiness level (0 - 255)
GUEST_STRUCT ENDS

MAX_GUESTS  EQU 1000           ; Simulate up to 1000 guests
GuestArray  GUEST_STRUCT MAX_GUESTS DUP (<>) ; Allocate array in memory
ParkCash    DD 500000          ; Global park treasury (e.g., $5000.00)

.CODE

; ------------------------------------------------------------------------------
; ROUTINE: UpdateGuestAndFinances
; PURPOSE: Loop through guests, decrease cash, increase park cash, increase nausea
; ------------------------------------------------------------------------------
UpdateGuestAndFinances PROC
    PUSH EBP                   ; Preserve base pointer
    MOV EBP, ESP               ; Set up stack frame
    PUSH ESI                   ; Preserve ESI (source index register)
    PUSH EBX                   ; Preserve EBX (base register)

    XOR ECX, ECX               ; ECX = 0 (Loop Counter / Guest Index)
    LEA ESI, GuestArray        ; ESI points to start of GuestArray

GuestLoop:
    CMP ECX, MAX_GUESTS        ; Have we checked every guest?
    JGE ExitLoop               ; If ECX >= 1000, exit loop

    ; Check if guest has money left for a ride ($2.00 or 200 cents)
    MOV BX, [ESI + GUEST_STRUCT.Cash]
    CMP BX, 200                
    JL NextGuest               ; Skip if guest has less than $2.00

    ; Deduct money from guest
    SUB BX, 200                
    MOV [ESI + GUEST_STRUCT.Cash], BX

    ; Add money to global park treasury
    ADD ParkCash, 200

    ; Increase guest Nausea level due to riding a coaster
    MOV AL, [ESI + GUEST_STRUCT.Nausea]
    ADD AL, 15                 ; Add nausea points
    JC MaxNausea               ; If addition overflows (>255), cap it
    MOV [ESI + GUEST_STRUCT.Nausea], AL
    JMP NextGuest

MaxNausea:
    MOV BYTE PTR [ESI + GUEST_STRUCT.Nausea], 255 ; Hard ceiling for nausea

NextGuest:
    ADD ESI, TYPE GUEST_STRUCT ; Advance pointer to next guest structure
    INC ECX                    ; Increment loop index counter
    JMP GuestLoop              ; Repeat loop

ExitLoop:
    POP EBX                    ; Restore registers
    POP ESI
    MOV ESP, EBP
    POP EBP
    RET                        ; Return to caller
UpdateGuestAndFinances ENDS
END

        GameActionResult FinalizeFlumeConstruction(const GameContext& context, const RideId rideId)
        {
            // 1. Fetch the strongly-typed Ride object from the manager (No mystery hex pointers)
            auto* ride = context.GetRideManager()->GetRide(rideId);
            if (ride == nullptr)
            {
                return GameActionResult::Failure(StringIds::InvalidRideError);

            // 2. Query the Track System to see if the track circuit closes back into itself
            TrackLayout track = context.GetTrackManager()->GetLayoutForRide(rideId);
            if (!track.IsCompleteCircuit())

                // Instead of a cryptic hex error code, return a localized system string
                return GameActionResult::Failure(StringIds::TrackIsAnIncompleteCircuit);
            }

            ride->SetStatusFlags(RideStatusFlags::CircuitComplete, true);

            context.GetMapRenderer()->InvalidateRideTracks(rideId);
    
            // 5. Play audio via a dedicated, abstracted Audio Context API

    void LogFlumeBoat::UpdatePhysics(const MapContext& map)
    {
        // Fetch the exact track piece the boat is sitting on right now
        TrackElement trackPiece = map.GetTrackElementAt(this->Position);
    
        if (trackPiece.IsWaterChannel())
        {
            // Apply constant forward propulsion from the flume's water pump current
            this->Velocity += WaterCurrentForce * TileDeltaTime;
            this->Velocity *= WaterFrictionDamping; // Water creates drag
        }
        
        if (trackPiece.IsDropSegment())
        {
            // Gravity takes over on a steep flume drop
            this->Velocity += GravityConstant * sin(trackPiece.SlopeAngle) * TileDeltaTime;
        }
    
        // Update the visual X, Y, Z coordinates based on velocity
        this->Position += this->Velocity * DirectionVector;
